 50    def validate(self, img: np.ndarray) -> bool:
 51        """Validates image dimensions, types, illumination, and structural plausibility.
 52
 53        Args:
 54            img: Input RGB image array.
 55
 56        Returns:
 57            bool: True if input strictly satisfies all validation constraints.
 58
 59        Raises:
 60            UnsupportedInputError: If any structural or radiometric check fails.
 61        """
 62        if not isinstance(img, np.ndarray):
 63            raise UnsupportedInputError(
 64                f"ERR_TYPE: Expected numpy.ndarray input, but received {type(img).__name__}."
 65            )
 66
 67        if img.shape != self.expected_shape:
 68            raise UnsupportedInputError(
 69                f"ERR_DIM: Incompatible dimensions {img.shape}. Expected {self.expected_shape} (Height, Width, Channels)."
 70            )
 71
 72        if img.dtype != np.uint8 and not np.issubdtype(img.dtype, np.floating):
 73            raise UnsupportedInputError(
 74                f"ERR_DTYPE: Unsupported array data type {img.dtype}. Must be uint8 or float32."
 75            )
 76
 77        mean_val = float(np.mean(img))
 78        var_val = float(np.var(img))
 79
 80        if mean_val < self.min_brightness:
 81            raise UnsupportedInputError(
 82                f"ERR_UNDEREXPOSED: Mean frame brightness ({mean_val:.1f}) is below minimum acceptable threshold ({self.min_brightness}). Possible camera occlusion or strobe trigger failure."
 83            )
 84
 85        if mean_val > self.max_brightness:
 86            raise UnsupportedInputError(
 87                f"ERR_OVEREXPOSED: Mean frame brightness ({mean_val:.1f}) exceeds maximum allowable threshold ({self.max_brightness}). Optical glare or severe overexposure."
 88            )
 89
 90        if var_val < self.min_variance:
 91            raise UnsupportedInputError(
 92                f"ERR_BLANK: Frame variance ({var_val:.1f}) is below {self.min_variance}. Frame lacks structural texture, indicating camera blackout or solid lens cap."
 93            )
 94
 95        # Domain Structural Check: Check if image has conveyor/tray structure
 96        # Tray border and conveyor background should have distinct spatial gradient distributions
 97        grad_y = np.abs(np.diff(img[:, :, 0].astype(np.float32), axis=0))
 98        mean_grad = float(np.mean(grad_y))
 99        
100        # Pure uncorrelated uniform noise has an unnaturally high mean gradient without macro edges
101        if mean_grad > 65.0:
102            raise UnsupportedInputError(
103                f"ERR_OUT_OF_DOMAIN: Mean spatial gradient ({mean_grad:.1f}) exceeds structured image bounds. Data appears to be uncorrelated noise or corrupted transmission."
104            )
105
106        return True

 40            height: Image height in pixels. Defaults to 540.

 47    def generate_sample(self, label: str) -> Tuple[np.ndarray, np.ndarray, bool]:
 48        """Generates a single synthetic 540x720x3 image, ground-truth defect mask, and label.
 49
 50        Args:
 51            label: Defect category ('pass', 'tear', 'seal_contam', 'discolor', 'unsupported').
 52
 53        Returns:
 54            Tuple containing:
 55                - img (np.ndarray): 540x720x3 RGB uint8 array.
 56                - mask (np.ndarray): 540x720 binary uint8 defect mask.
 57                - is_defective (bool): True if sample represents a packaging or quality failure.
 58        """
 59        img = np.zeros((self.h, self.w, 3), dtype=np.uint8)
 60        mask = np.zeros((self.h, self.w), dtype=np.uint8)
 61
 62        if label == "unsupported":
 63            if self.rng.rand() > 0.5:
 64                # Blackout sensor failure
 65                return img, mask, False
 66            else:
 67                # Uncorrelated uniform random noise
 68                noisy = self.rng.randint(0, 256, (self.h, self.w, 3), dtype=np.uint8)
 69                return noisy, mask, False
 70
 71        # Conveyor belt matte finish
 72        img[:, :] = [45, 48, 52]
 73
 74        # Thermoformed white polymer tray perimeter
 75        ty1, tx1, ty2, tx2 = 60, 80, 480, 640
 76        img[ty1:ty2, tx1:tx2] = [212, 216, 222]
 77
 78        # Inner tray cavity
 79        cy1, cx1, cy2, cx2 = 80, 100, 460, 620
 80        img[cy1:cy2, cx1:cx2] = [172, 176, 182]
 81
 82        # Packaged pork rashers with alternating lean and intramuscular fat striations
 83        rasher_y_starts = [100, 170, 240, 310, 380]
 84        for y0 in rasher_y_starts:
 85            # Lean muscle (oxymyoglobin red)
 86            img[y0 : y0 + 45, 120:600] = [192, 72, 82]
 87            # Intramuscular fat stripe
 88            img[y0 + 15 : y0 + 26, 120:600] = [232, 222, 208]
 89
 90        # Add Gaussian sensor noise
 91        noise = self.rng.normal(0, 3.5, (self.h, self.w, 3)).astype(np.int16)
 92        img = np.clip(img.astype(np.int16) + noise, 0, 255).astype(np.uint8)
 93
 94        is_defective = False
 95
 96        if label == "tear":
 97            py = self.rng.randint(150, 380)
 98            px = self.rng.randint(180, 500)
 99            radius = self.rng.randint(18, 36)
100            yy, xx = np.ogrid[: self.h, : self.w]
101            dist = np.sqrt((xx - px) ** 2 + (yy - py) ** 2)
102            defect_region = dist <= radius
103            img[defect_region] = [22, 22, 28]  # Dark puncture / rupture
104            mask[defect_region] = 1
105            is_defective = True
106
107        elif label == "seal_contam":
108            sy = self.rng.randint(ty1, ty1 + 18)
109            sx = self.rng.randint(tx1 + 60, tx2 - 120)
110            img[sy : sy + 14, sx : sx + 45] = [168, 52, 62]  # Trapped meat in seal
111            mask[sy : sy + 14, sx : sx + 45] = 1
112            is_defective = True
113
114        elif label == "discolor":
115            dy = self.rng.randint(120, 380)
116            dx = self.rng.randint(150, 520)
117            yy, xx = np.ogrid[: self.h, : self.w]
118            dist = np.sqrt((xx - dx) ** 2 + (yy - dy) ** 2)
119            discolor_region = dist <= 32
120            img[discolor_region] = [108, 98, 78]  # Metmyoglobin brown oxidation
121            mask[discolor_region] = 1
122            is_defective = True
123
124        return img, mask, is_defective
